cmake_minimum_required(VERSION 3.20)
project(rowdogg VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
# Pinned, header-only libraries keep the runtime entirely in C++.
FetchContent_Declare(crow URL https://github.com/CrowCpp/Crow/archive/refs/tags/v1.2.1.tar.gz URL_HASH SHA256=552f2e447adf70ed4c667d6f82db53dfc70710b50431004ab1405f5b53f04c30)
FetchContent_Declare(asio URL https://github.com/chriskohlhoff/asio/archive/refs/tags/asio-1-30-2.tar.gz URL_HASH SHA256=755bd7f85a4b269c67ae0ea254907c078d408cce8e1a352ad2ed664d233780e8)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.1.tar.gz URL_HASH SHA256=405abd8170f2a446fc8612ac635d0db5947c0d2e156e32603403a4496255ff00)
foreach(dependency crow asio json httplib)
  FetchContent_GetProperties(${dependency})
  if(NOT ${dependency}_POPULATED)
    FetchContent_Populate(${dependency})
  endif()
endforeach()
find_package(Threads REQUIRED)
add_library(rowdogg_core backend/src/solvers.cpp backend/src/engine.cpp backend/src/gemma.cpp)
target_include_directories(rowdogg_core PUBLIC backend/include ${json_SOURCE_DIR}/include ${httplib_SOURCE_DIR})
target_link_libraries(rowdogg_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(rowdogg_core PUBLIC ws2_32 mswsock crypt32)
  target_compile_definitions(rowdogg_core PUBLIC _WIN32_WINNT=0x0A00)
endif()
add_executable(rowdogg backend/src/main.cpp)
target_include_directories(rowdogg PRIVATE ${crow_SOURCE_DIR}/include ${asio_SOURCE_DIR}/asio/include)
target_compile_definitions(rowdogg PRIVATE ASIO_STANDALONE CROW_ENABLE_DEBUG=0)
target_link_libraries(rowdogg PRIVATE rowdogg_core)
enable_testing()
add_executable(engine_tests backend/tests/engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE rowdogg_core)
add_test(NAME engine_tests COMMAND engine_tests)
add_executable(benchmark_runner backend/tests/benchmark.cpp)
target_link_libraries(benchmark_runner PRIVATE rowdogg_core)
add_test(NAME benchmark COMMAND benchmark_runner ${CMAKE_SOURCE_DIR}/data/benchmark.json)
